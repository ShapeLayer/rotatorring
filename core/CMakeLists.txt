cmake_minimum_required(VERSION 3.20)
project(RotatorringCore VERSION 1.0.0 LANGUAGES C)
include(CTest)
add_library(RotatorringCore SHARED src/rotatorring_core.c)
target_include_directories(RotatorringCore PUBLIC include)
target_compile_features(RotatorringCore PRIVATE c_std_11)
target_compile_definitions(RotatorringCore PUBLIC RR_SHARED PRIVATE RR_BUILD)
if(MSVC)
  set_property(TARGET RotatorringCore PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(RotatorringCore PRIVATE /W4 /WX)
else()
  target_compile_options(RotatorringCore PRIVATE -Wall -Wextra -Werror -pedantic)
  target_link_libraries(RotatorringCore PRIVATE m)
endif()
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.c)
  target_link_libraries(core_tests PRIVATE RotatorringCore)
  if(NOT MSVC)
    target_link_libraries(core_tests PRIVATE m)
  endif()
  add_test(NAME core_geometry COMMAND core_tests)
endif()
